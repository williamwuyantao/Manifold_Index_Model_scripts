  declare function={
    eigsat(\r,\L,\s)=\L*(\r/\s)^2/sqrt(1+(\r/\s)^4)*(1+.012*\r);
    eignormal(\r)=eigsat(\r,72,32);
    eigtangent(\r)=eigsat(\r,5.57,12);
    eigstep(\u)=(min(1,max(0,\u)))^3
      *(10-15*min(1,max(0,\u))+6*(min(1,max(0,\u)))^2);
    eigjoin(\r)=eigtangent(\r)+150*eigstep((\r-46)/30)*(1+.012*\r);
    eigsoft(\a,\b)=ln(1+exp(-6*abs(ln(\a)-ln(\b))))/6;
    eigfirst(\r)=max(ln(eignormal(\r)),ln(eigjoin(\r)))
      +eigsoft(eignormal(\r),eigjoin(\r));
    eiggap(\r)=min(ln(eigjoin(\r)),ln(.43*eignormal(\r)))
      -eigsoft(eigjoin(\r),.43*eignormal(\r));
    eigscore(\r)=2*(ln(eignormal(\r))-.3993)-eigfirst(\r)-eiggap(\r);
    eigy(\l)=13*(\l+1.9);
  }]
